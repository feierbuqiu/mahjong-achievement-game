// Untrusted candidate production only. The existing database is opened read-only.
// Protocol: E <34 counts> -> physical SAFE edges; Q <34 counts> -> root metadata.
// Every emitted symmetry path is replayed here, and must later be checked in Lean.

namespace producer {
using oq::require;
constexpr const char* SCHEMA="MAHJONG_STRATEGY_WORKER_V1";
struct Generator {int op,a,b;};
void apply_generator(std::string& s,const Generator& g){
  if(g.op==0)std::reverse(s.begin()+9*g.a,s.begin()+9*(g.a+1));
  else if(g.op==1)for(int i=0;i<9;++i)std::swap(s[9*g.a+i],s[9*g.b+i]);
  else std::swap(s[27+g.a],s[27+g.b]);
}
std::string path_json(const std::vector<Generator>& path){
  std::ostringstream o;o<<'[';bool first=true;
  for(auto&g:path){if(!first)o<<',';first=false;
    if(g.op==0)o<<"{\"op\":\"reverseSuit\",\"suit\":"<<g.a<<'}';
    else o<<"{\"op\":\""<<(g.op==1?"swapSuits":"swapHonors")<<"\",\"a\":"<<g.a<<",\"b\":"<<g.b<<'}';
  }o<<']';return o.str();
}
struct Canonical {std::string text;std::vector<Generator> path;};
Canonical canonical_path(const std::string& raw){
  std::string s=raw;std::vector<Generator> forward;
  auto perform=[&](Generator g){apply_generator(s,g);forward.push_back(g);};
  for(int b=0;b<3;++b){auto x=s.substr(9*b,9),r=x;std::reverse(r.begin(),r.end());if(r<x)perform({0,b,0});}
  for(int i=0;i<3;++i){int least=i;for(int j=i+1;j<3;++j)if(s.substr(9*j,9)<s.substr(9*least,9))least=j;if(least!=i)perform({1,i,least});}
  for(int i=0;i<7;++i){int least=i;for(int j=i+1;j<7;++j)if(s[27+j]<s[27+least])least=j;if(least!=i)perform({2,i,least});}
  require(s==::repr(::canon(::parse(raw))),"explicit symmetry path disagrees with canonicalizer");
  std::reverse(forward.begin(),forward.end());std::string replay=s;for(auto&g:forward)apply_generator(replay,g);
  require(replay==raw,"symmetry witness does not reconstruct physical state");return {s,forward};
}
void strict_digits(const std::string&s){require(s.size()==34&&s.find_first_not_of("01234")==std::string::npos,"expected exactly 34 digits 0..4");}
struct Value {bool p;int n,h;uint64_t rank;};
class Worker {
  oq::Database& db;
  // Bounded full-key cache; clear-all eviction changes speed, never decisions.
  std::unordered_map<State,Value,Hash> cache;
  std::unordered_map<State,unsigned char,Hash> degrees;
  std::mutex value_mutex,degree_mutex;
public:
  uint64_t lookups=0,hits=0;
  std::atomic<uint64_t> requests{0};
  explicit Worker(oq::Database& d):db(d){cache.reserve(500000);degrees.reserve(250000);}
  Value lookup(State s){
    std::lock_guard<std::mutex> guard(value_mutex);
    s=::canon(s);auto it=cache.find(s);if(it!=cache.end()){++hits;return it->second;}
    auto x=db.lookup(s);++lookups;require(x.known,"unknown database state: "+x.reason);
    Value v{x.p,x.n,x.h,x.rank};if(cache.size()>=1000000)cache.clear();cache.emplace(s,v);return v;
  }
  int degree(const State&s){
    {std::lock_guard<std::mutex> guard(degree_mutex);auto it=degrees.find(s);if(it!=degrees.end())return it->second;}
    // Do not use ::moves here: its legacy generated counter is mutable global state.
    std::vector<State> unique;
    for(int b=0;b<4;++b){uint32_t r=s.c[b];int width=b==3?7:9;
      for(int i=0;i<width;++i){int d=r%5;r/=5;if(d==4)continue;State t=s;t.c[b]+=::pw[i];
        if(!::safe(t))continue;t=::canon(t);if(std::find(unique.begin(),unique.end(),t)==unique.end())unique.push_back(t);}}
    int d=int(unique.size());std::lock_guard<std::mutex> guard(degree_mutex);
    if(degrees.size()>=500000)degrees.clear();degrees.emplace(s,d);return d;
  }
  std::string request(char command,const std::string& text){
    db.check_budget();strict_digits(text);auto state=::parse(text);require(::safe(state),"root/node is not SAFE");
    const auto canonical=canonical_path(text);auto parent=lookup(state);++requests;
    std::ostringstream out;out<<"{\"schema\":"<<oq::quote(SCHEMA)<<",\"kind\":\""<<(command=='Q'?"root":"expansion")
      <<"\",\"input_state\":"<<oq::quote(text)<<",\"canonical_state\":"<<oq::quote(canonical.text)
      <<",\"root_witness\":"<<path_json(canonical.path)<<",\"outcome\":\""<<(parent.p?'P':'N')<<"\",\"n\":"<<parent.n
      <<",\"h\":"<<parent.h<<",\"rank\":"<<parent.rank;
    int physical=0,retained=0;out<<",\"edges\":[";
    if(command=='E')for(int tile=0;tile<34;++tile)if(text[tile]<'4'){
      db.check_budget();auto raw=text;++raw[tile];auto child=::parse(raw);if(!::safe(child))continue;++physical;
      child=::canon(child);auto value=lookup(child);
      require(!parent.p||!value.p,"P database node has a P SAFE child");
      if(!parent.p&&!value.p)continue;
      auto c=canonical_path(raw);if(retained++)out<<',';
      out<<"{\"tile\":"<<tile<<",\"raw_successor\":"<<oq::quote(raw)<<",\"canonical_successor\":"<<oq::quote(c.text)
        <<",\"outcome\":\""<<(value.p?'P':'N')<<"\",\"witness\":"<<oq::quote(path_json(c.path))
        <<",\"future_orbit_degree\":"<<(value.p?degree(child):-1)<<'}';
    }
    if(command=='E'&&!parent.p)require(retained>0,"N database node has no SAFE P reply");
    std::lock_guard<std::mutex> guard(value_mutex);
    out<<"],\"physical_safe_moves\":"<<physical<<",\"retained_candidates\":"<<retained
      <<",\"requests\":"<<requests.load()<<",\"database_lookups\":"<<lookups<<",\"cache_hits\":"<<hits
      <<",\"verified_payload_bytes\":"<<db.verified_payload_bytes<<'}';return out.str();
  }
};
}

int main(int argc,char**argv){try{
  oq::require(argc==5,"usage: strategy_worker_v2 DATA SECONDS EXPECTED_FINGERPRINT THREADS");
  int threads=std::stoi(argv[4]);oq::require(threads>=1&&threads<=16,"thread count outside 1..16");
  double seconds=std::stod(argv[2]);oq::require(seconds>0,"positive time budget required");
  std::ios::sync_with_stdio(false);std::cin.tie(nullptr);
  const auto start=Clock::now();retro::Index index;oq::require(oq::hex(index.fingerprint)==argv[3],"index fingerprint mismatch");
  oq::Database db(index,{std::filesystem::u8path(argv[1])},seconds);producer::Worker worker(db);
  std::cout<<"{\"kind\":\"ready\",\"schema\":\"MAHJONG_STRATEGY_WORKER_V1\",\"index_fingerprint\":\""<<oq::hex(index.fingerprint)
    <<"\",\"setup_seconds\":"<<std::chrono::duration<double>(Clock::now()-start).count()<<'}'<<std::endl;
  struct Task {uint64_t id;std::string line;};
  std::mutex input_mutex,output_mutex;std::condition_variable input_cv,output_cv;
  std::deque<Task> tasks;std::map<uint64_t,std::string> results;bool input_done=false;int finished=0;
  std::vector<std::thread> pool;
  for(int i=0;i<threads;++i)pool.emplace_back([&]{
    for(;;){Task task;
      {std::unique_lock<std::mutex> lock(input_mutex);input_cv.wait(lock,[&]{return input_done||!tasks.empty();});
        if(tasks.empty())break;task=std::move(tasks.front());tasks.pop_front();}
      std::string result;
      try{
        const auto&line=task.line;oq::require(line.size()==36&&(line[0]=='E'||line[0]=='Q')&&line[1]==' ',"invalid worker command");
        result=worker.request(line[0],line.substr(2));
      }catch(const oq::Missing&e){result="{\"kind\":\"bounded_stop\",\"reason\":"+oq::quote(e.what())+'}';}
      catch(const std::exception&e){result="{\"kind\":\"error\",\"reason\":"+oq::quote(e.what())+'}';}
      {std::lock_guard<std::mutex> lock(output_mutex);results.emplace(task.id,std::move(result));}output_cv.notify_one();
    }
    {std::lock_guard<std::mutex> lock(output_mutex);++finished;}output_cv.notify_one();
  });
  std::thread writer([&]{uint64_t next=0;for(;;){std::string result;
    {std::unique_lock<std::mutex> lock(output_mutex);output_cv.wait(lock,[&]{return results.count(next)||finished==threads;});
      auto it=results.find(next);if(it==results.end())break;result=std::move(it->second);results.erase(it);++next;}
    std::cout<<result<<'\n';std::cout.flush();
  }});
  std::string line;uint64_t ordinal=0;
  while(std::getline(std::cin,line)){
    if(!line.empty()&&line.back()=='\r')line.pop_back();if(line=="STOP")break;
    {std::lock_guard<std::mutex> lock(input_mutex);tasks.push_back({ordinal++,std::move(line)});}input_cv.notify_one();
  }
  {std::lock_guard<std::mutex> lock(input_mutex);input_done=true;}input_cv.notify_all();
  for(auto&t:pool)t.join();writer.join();
  return 0;
}catch(const oq::Missing&e){std::cout<<"{\"kind\":\"bounded_stop\",\"reason\":"<<oq::quote(e.what())<<'}'<<std::endl;return 2;}
catch(const std::exception&e){std::cout<<"{\"kind\":\"error\",\"reason\":"<<oq::quote(e.what())<<'}'<<std::endl;return 1;}}

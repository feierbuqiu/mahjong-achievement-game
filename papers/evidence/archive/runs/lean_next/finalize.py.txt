HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]


def digest(path):
    with Path(path).open("rb") as source:
        return hashlib.file_digest(source, "sha256").hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8-sig"))


def receipt(path):
    path = Path(path)
    return {"path": path.relative_to(ROOT).as_posix(), "bytes": path.stat().st_size, "sha256": digest(path)}


def check_clean(name):
    folder = HERE / name
    result = read(folder / "result.json")
    assert result["kernel_modules"] == "PASS"
    assert result["trust_level"] == 0 and result["empty_game"] == "NOT_VERIFIED"
    assert result["passed_modules"] == result["total_modules"] == len(result["steps"])
    assert result["source_unchanged"]
    assert digest(folder / "runner.py") == result["runner_sha256"]
    assert digest(ROOT / "build/toolchain/lean-4.33.1-windows/bin/lean.exe") == result["toolchain_sha256"]
    for path, expected in result["source_sha256"].items():
        assert digest(folder / "source" / path) == expected, path
    for path, expected in result["artifacts_sha256"].items():
        assert digest(folder / "modules" / path) == expected, path
    for step in result["steps"]:
        assert step["return_code"] == 0 and step["stop_reason"] is None
        assert "--trust=0" in step["command"]
        assert digest(step["log"]) == step["log_sha256"]
        if "reused_from" in step:
            assert digest(Path(step["reused_from"]) / "result.json") == result["reused_base_receipt_sha256"]
    log = (folder / (result["entry"] + ".log")).read_text(encoding="utf-8-sig")
    assert "AXIOM_AUDIT_PASS" in log and "END_TO_END_NOT_VERIFIED" in log
    assert "END_TO_END_CLOSED_THEOREM_PRESENT" not in log
    return result


def stopped_progress(folder):
    with (folder / "run.lock").open("r+b") as handle:
        handle.seek(0)
        msvcrt.locking(handle.fileno(), msvcrt.LK_NBLCK, 1)
        try:
            return read(folder / "progress.latest.json")
        finally:
            handle.seek(0)
            msvcrt.locking(handle.fileno(), msvcrt.LK_UNLCK, 1)

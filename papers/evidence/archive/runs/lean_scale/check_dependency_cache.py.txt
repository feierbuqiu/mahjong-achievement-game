"""A changed imported definition must invalidate a same-source dependent proof."""
from pathlib import Path
import hashlib
import json
import subprocess
import sys

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def run(name, reuse=None):
    args = [sys.executable, str(HERE / "kernel_audit_v2.py"), "--entry", "ScaleCacheProbeAudit",
            "--output", name, "--workers", "2", "--memory-budget", "4000000000", "--timeout", "60"]
    if reuse:
        args += ["--reuse", "runs/lean_scale/" + reuse]
    done = subprocess.run(args, cwd=ROOT, creationflags=subprocess.CREATE_NO_WINDOW,
                          stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=60)
    (HERE / name / "driver.log").write_bytes(done.stdout)
    return json.loads((HERE / name / "result.json").read_text(encoding="utf-8"))


def main():
    base = ROOT / "build/lean_scale/proof/ScaleCacheProbe/Base.lean"
    original = base.read_bytes()
    assert b"value : Nat := 1" in original
    proof = ROOT / "build/lean_scale/proof/ScaleCacheProbe/Proof.lean"
    before_proof = sha(proof)
    try:
        first = run("cache_baseline_v2")
        assert first["kernel_modules"] == "PASS"
        base.write_bytes(original.replace(b"value : Nat := 1", b"value : Nat := 2"))
        second = run("cache_changed_dependency_v2", "cache_baseline_v2")
        assert second["kernel_modules"] != "PASS"
        assert sha(proof) == before_proof
        step = next(s for s in second["steps"] if s["module"] == "ScaleCacheProbe.Proof")
        assert step["return_code"] != 0 and "reused_from" not in step
        error = Path(step["log"]).read_text(encoding="utf-8-sig")
        assert "error" in error and "rfl" in error
    finally:
        base.write_bytes(original)
    out = {"schema": "DEPENDENCY_EXACT_CACHE_REGRESSION_V1", "status": "PASS_EXPECTED_REJECTION",
        "baseline": "KERNEL_PASS", "changed_dependency": "RECOMPILED_AND_REJECTED",
        "dependent_source_unchanged": sha(proof) == before_proof,
        "fixture_source_restored": base.read_bytes() == original,
        "Mahjong_conclusion": "NOT_ASSERTED_BY_FIXTURE",
        "sha256": {p.relative_to(ROOT).as_posix(): sha(p) for p in [Path(__file__), HERE / "kernel_audit_v2.py",
            HERE / "cache_baseline_v2/result.json", HERE / "cache_changed_dependency_v2/result.json"]}}
    with (HERE / "dependency_cache_regression.json").open("x", encoding="utf-8") as stream:
        json.dump(out, stream, indent=2)
        stream.write("\n")
    print(json.dumps(out))


if __name__ == "__main__":
    main()
